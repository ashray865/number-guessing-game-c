main.c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main()
{
    srand(time(0));

    int randomNumber = (rand() % 100) + 1;
    int guessed;
    int no_of_guesses = 0;

    do
    {
        scanf("%d", &guessed);
        no_of_guesses++;

        if (guessed > randomNumber)
            printf("Lower number please\n");
        else if (guessed < randomNumber)
            printf("Higher number please\n");
        else
            printf("Congratulations!");

    } while (guessed != randomNumber);

    return 0;
}
